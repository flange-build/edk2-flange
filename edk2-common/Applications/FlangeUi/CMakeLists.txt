cmake_minimum_required(VERSION 3.16)
project(FlangeUiPreview C)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../Include/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../ThirdParty/lvgl lvgl)
add_executable(flange-preview Tools/preview.c FlangeValue.c FlangeTheme.c FlangePanel.c FlangeView.c FlangeText.c Fonts/FlangeBody20.c Fonts/FlangeCjk24.c Fonts/FlangeSmall18.c Fonts/FlangeLabel14.c Fonts/FlangeMono14.c Fonts/FlangeMono20.c Fonts/FlangeTitle28.c)
target_include_directories(flange-preview PRIVATE . ../../ThirdParty ../../Include)
target_link_libraries(flange-preview PRIVATE lvgl)
if(FLANGE_SANITIZE)
  target_compile_options(lvgl PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(flange-preview PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(flange-preview PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME view-1080p COMMAND flange-preview 1920 1080 zh preview-1080p.ppm)
add_test(NAME view-768p COMMAND flange-preview 1024 768 en preview-768p.ppm)
add_test(NAME view-600p COMMAND flange-preview 800 600 zh preview-600p.ppm)
add_test(NAME view-empty COMMAND flange-preview 800 600 zh preview-empty.ppm 0)
add_test(NAME view-long-options COMMAND flange-preview 800 600 en preview-long.ppm 19)
